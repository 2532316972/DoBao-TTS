## 1. 前置条件

你的项目需要满足：

| 条件 | 说明 |
|------|------|
| Node.js + Express | Web 框架 |
| SQLite | 数据库（使用 `sqlite` + `sqlite3` 包） |
| 已有 logger 工具 | 例如 `require('../utils/logger')` 中的 `logger` |
| Gitee 仓库 | 同一个共享仓库 `AnnouncementManager`，你在里面有自己的子目录 |

> **如果项目没有 SQLite**：可以改用 JSON 文件做本地存储，但建议先装好 SQLite。
>
> ```bash
> npm install sqlite sqlite3
> ```

### 你需要修改的唯一配置项

在核心服务文件中，`_GITEE_CONFIG` 对象只有 **`path`** 需要改：

```js
const _GITEE_CONFIG = Object.freeze({
```

## 2. 第一步：创建核心服务文件

在 `src/services/` 目录下创建 **`userStatsService.js`**，完整内容如下：

```js
const crypto = require('crypto');
const { logger } = require('../utils/logger');

const _GITEE_CONFIG = Object.freeze({
  token: process.env.GITEE_SYNC_TOKEN || '',  // 令牌从环境变量注入，不要写死
  owner: process.env.GITEE_SYNC_OWNER || '你的Gitee用户名',
  repo: 'AnnouncementManager',
  path: '你的项目名/user-stats.json',
  branch: 'master'
});

function getSafeConfig() {
  return {
    ..._GITEE_CONFIG,
    token: _GITEE_CONFIG.token ? '***已配置***' : '未配置'
  };
}

let _now = () => new Date();

function setNow(fn) {
  _now = fn;
}

function getToday() {
  return _now().toISOString().split('T')[0];
}

let _db = null;
let _testDb = null;

function setTestDb(db) {
  _testDb = db;
  _db = null;
}

async function getDb() {
  if (_testDb) {
    await initTables(_testDb);
    return _testDb;
  }
  if (!_db) {
    const { getDb: getMainDb } = require('../db/sqlite');
    _db = await getMainDb();
    await initTables(_db);
  }
  return _db;
}

async function initTables(db) {
  await db.exec(`
    CREATE TABLE IF NOT EXISTS user_access_log (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      user_hash TEXT NOT NULL,
      ip TEXT,
      user_agent TEXT,
      accessed_at DATETIME DEFAULT CURRENT_TIMESTAMP
    );
    CREATE INDEX IF NOT EXISTS idx_access_log_hash ON user_access_log(user_hash);
    CREATE INDEX IF NOT EXISTS idx_access_log_date ON user_access_log(accessed_at);
  `);

  await db.exec(`
    CREATE TABLE IF NOT EXISTS daily_user_stats (
      date TEXT PRIMARY KEY,
      user_count INTEGER DEFAULT 0,
      updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
    );
  `);
}

function normalizeIp(ip) {
  if (!ip) return 'unknown';
  if (ip === '127.0.0.1' || ip === '::1' || ip === '::ffff:127.0.0.1') {
    return 'local';
  }
  if (ip.startsWith('192.168.') ||
      ip.startsWith('10.') ||
      ip.startsWith('::ffff:192.168.') ||
      ip.startsWith('::ffff:10.')) {
    return 'local';
  }
  const match = ip.match(/^(?:::ffff:)?172\.(1[6-9]|2[0-9]|3[01])\./);
  if (match) {
    return 'local';
  }
  return ip;
}

function generateUserHash(ip, userAgent) {
  const normalizedIp = normalizeIp(ip);
  const raw = `${normalizedIp}|${userAgent || 'unknown'}`;
  return crypto.createHash('md5').update(raw).digest('hex');
}

async function recordAccess({ ip, userAgent }) {
  try {
    const db = await getDb();
    const userHash = generateUserHash(ip, userAgent);
    const today = getToday();
    const currentTime = _now().toISOString();
    const dateTimeForSqlite = today + ' ' + currentTime.split('T')[1].split('.')[0];

    await db.run(
      `INSERT INTO user_access_log (user_hash, ip, user_agent, accessed_at)
       VALUES (?, ?, ?, ?)`,
      [userHash, ip || '', userAgent || '', dateTimeForSqlite]
    );

    return { success: true, userHash };
  } catch (error) {
    logger.error('[UserStats] 记录访问失败', { error: error.message });
    return { success: false, error: error.message };
  }
}

async function getDailyStats() {
  try {
    const db = await getDb();
    const today = getToday();

    const result = await db.get(
      `SELECT COUNT(DISTINCT user_hash) as count 
       FROM user_access_log 
       WHERE date(accessed_at) = ?`,
      [today]
    );

    return {
      date: today,
      todayCount: result?.count || 0,
      updatedAt: new Date().toISOString()
    };
  } catch (error) {
    logger.error('[UserStats] 获取统计失败', { error: error.message });
    return {
      date: getToday(),
      todayCount: 0,
      error: error.message
    };
  }
}

async function updateDailyStats() {
  try {
    const db = await getDb();
    const stats = await getDailyStats();

    await db.run(
      `INSERT INTO daily_user_stats (date, user_count, updated_at) 
       VALUES (?, ?, datetime('now'))
       ON CONFLICT(date) DO UPDATE SET 
         user_count = excluded.user_count, 
         updated_at = excluded.updated_at`,
      [stats.date, stats.todayCount]
    );

    logger.info('[UserStats] 统计已更新', { date: stats.date, count: stats.todayCount });
    return stats;
  } catch (error) {
    logger.error('[UserStats] 更新统计失败', { error: error.message });
    return { error: error.message };
  }
}

function startScheduler(options = {}) {
  const {
    intervalMs = 2 * 60 * 60 * 1000,
    enableGiteeSync = true
  } = typeof options === 'number' ? { intervalMs: options } : options;

  logger.info('[UserStats] 启动定时统计任务', {
    interval: `${intervalMs / 1000 / 60}分钟`,
    giteeSync: enableGiteeSync ? '启用' : '禁用',
    config: getSafeConfig()
  });

  async function runStats() {
    try {
      await updateDailyStats();
      if (enableGiteeSync) {
        await syncToGitee();
      }
    } catch (error) {
      logger.error('[UserStats] 定时任务执行失败', { error: error.message });
    }
  }

  runStats();
  const timer = setInterval(runStats, intervalMs);

  return {
    stop: () => {
      clearInterval(timer);
      logger.info('[UserStats] 定时统计任务已停止');
    }
  };
}

async function generateSyncData() {
  try {
    const db = await getDb();
    const today = getToday();
    const stats = await getDailyStats();

    const historicalStats = await db.all(
      `SELECT date, user_count as userCount, updated_at as updatedAt 
       FROM daily_user_stats 
       ORDER BY date DESC 
       LIMIT 30`
    );

    return {
      version: '1.0.0',
      lastUpdated: new Date().toISOString(),
      today: {
        date: today,
        userCount: stats.todayCount
      },
      dailyStats: historicalStats.length > 0 ? historicalStats : [{
        date: today,
        userCount: stats.todayCount,
        updatedAt: stats.updatedAt
      }]
    };
  } catch (error) {
    logger.error('[UserStats] 生成同步数据失败', { error: error.message });
    return {
      version: '1.0.0',
      lastUpdated: new Date().toISOString(),
      today: { date: getToday(), userCount: 0 },
      dailyStats: [],
      error: error.message
    };
  }
}

async function syncToGitee() {
  const { token, owner, repo, path, branch } = _GITEE_CONFIG;

  if (!token) {
    logger.warn('[UserStats] Gitee token未配置，跳过同步');
    return { success: false, error: 'token未配置' };
  }

  try {
    const syncData = await generateSyncData();
    const content = Buffer.from(JSON.stringify(syncData, null, 2)).toString('base64');

    let sha = '';
    let fileExists = false;
    try {
      const existingFile = await giteeRequest(
        `/repos/${owner}/${repo}/contents/${path}`, 'GET', null, token
      );
      if (existingFile.sha) {
        sha = existingFile.sha;
        fileExists = true;
      }
    } catch (e) {
      // 文件不存在，首次创建
    }

    const body = {
      message: `更新用户统计 - ${new Date().toISOString()}`,
      content,
      branch
    };
    if (sha) body.sha = sha;

    const method = fileExists ? 'PUT' : 'POST';
    const result = await giteeRequest(
      `/repos/${owner}/${repo}/contents/${path}`, method, body, token
    );

    logger.info('[UserStats] 数据已同步到Gitee');
    return { success: true, url: result.content?.html_url };
  } catch (error) {
    logger.error('[UserStats] 同步到Gitee失败', { error: error.message });
    return { success: false, error: error.message };
  }
}

function giteeRequest(apiPath, method, body, token) {
  const https = require('https');
  const separator = apiPath.includes('?') ? '&' : '?';
  const fullPath = token ? `${apiPath}${separator}access_token=${token}` : apiPath;

  return new Promise((resolve, reject) => {
    const options = {
      hostname: 'gitee.com',
      path: `/api/v5${fullPath}`,
      method,
      headers: {
        'Content-Type': 'application/json',
        'User-Agent': 'YourProject-UserStats'
      }
    };

    const req = https.request(options, (res) => {
      let data = '';
      res.on('data', chunk => data += chunk);
      res.on('end', () => {
        try {
          const parsed = JSON.parse(data);
          if (res.statusCode >= 200 && res.statusCode < 300) {
            resolve(parsed);
          } else {
            logger.error('[Gitee] API请求失败', {
              statusCode: res.statusCode,
              path: options.path,
              method,
              response: data.substring(0, 1000)
            });
            reject(new Error(parsed.message || parsed.error || `HTTP ${res.statusCode}`));
          }
        } catch (e) {
          reject(new Error('解析响应失败'));
        }
      });
    });

    req.on('error', reject);
    req.setTimeout(15000, () => {
      req.destroy();
      reject(new Error('请求超时'));
    });

    if (body) req.write(JSON.stringify(body));
    req.end();
  });
}

module.exports = {
  setTestDb,
  setNow,
  getToday,
  recordAccess,
  getDailyStats,
  updateDailyStats,
  startScheduler,
  generateUserHash,
  generateSyncData,
  syncToGitee
};
```

## 3. 第二步：创建 API 路由文件

在 `src/routes/` 目录下创建 **`stats.js`**：

```js
const express = require('express');
const router = express.Router();
const { logger } = require('../utils/logger');
const {
  getDailyStats,
  recordAccess,
  generateSyncData,
  syncToGitee,
  updateDailyStats
} = require('../services/userStatsService');

router.use((req, res, next) => {
  const ip = req.ip || req.connection?.remoteAddress || '';
  const userAgent = req.get('user-agent') || '';

  recordAccess({ ip, userAgent }).catch(err => {
    logger.debug('[Stats] 记录访问失败:', err.message);
  });

  next();
});

router.get('/daily', async (req, res) => {
  try {
    const stats = await getDailyStats();
    res.json({ success: true, data: stats });
  } catch (error) {
    logger.error('[Stats] 获取统计失败:', error.message);
    res.status(500).json({ success: false, error: '获取统计数据失败' });
  }
});

router.get('/sync-data', async (req, res) => {
  try {
    const data = await generateSyncData();
    res.json({ success: true, data });
  } catch (error) {
    logger.error('[Stats] 获取同步数据失败:', error.message);
    res.status(500).json({ success: false, error: '获取同步数据失败' });
  }
});

router.post('/update', async (req, res) => {
  try {
    const stats = await updateDailyStats();
    res.json({ success: true, message: '统计已更新', data: stats });
  } catch (error) {
    logger.error('[Stats] 更新统计失败:', error.message);
    res.status(500).json({ success: false, error: '更新统计失败' });
  }
});

router.post('/sync', async (req, res) => {
  try {
    const syncResult = await syncToGitee();
    if (syncResult.success) {
      res.json({ success: true, message: '数据已同步到Gitee', url: syncResult.url });
    } else {
      res.status(500).json({ success: false, error: syncResult.error });
    }
  } catch (error) {
    logger.error('[Stats] Gitee同步失败:', error.message);
    res.status(500).json({ success: false, error: 'Gitee同步失败' });
  }
});

router.get('/health', (req, res) => {
  res.json({ success: true, service: 'user-stats', timestamp: new Date().toISOString() });
});

module.exports = router;
```

此文件 **无需任何修改**，直接复制即可。

## 4. 第三步：在主入口注册路由 + 启动定时任务

打开你的 **`src/index.js`**（或 `app.js`，即 Express 入口文件），添加以下两处：

### 4.1 引入并注册路由

在文件顶部的 `require` 区域添加：

```js
const statsRouter = require('./routes/stats');
```

在 `app.use(...)` 路由注册区域添加：

```js
app.use('/api/stats', statsRouter);
```

> 放在其他路由注册旁边即可，位置不限。

### 4.2 启动定时统计任务

在 `app.listen(...)` 之后（或服务器启动成功的回调中）添加：

```js
try {
  const { startScheduler } = require('./services/userStatsService');
  startScheduler(2 * 60 * 60 * 1000); // 每2小时执行一次
  logger.info('[Server] 用户统计定时任务已启动');
} catch (err) {
  logger.error('[Server] 启动用户统计任务失败:', err.message);
}
```

> 如果项目没有全局 `logger`，可以替换为 `console.log`/`console.error`。

## Gitee 同步原理说明

### 数据流

```
本地 SQLite (user_access_log)
       │
       ▼ 每2小时聚合
本地 SQLite (daily_user_stats)
       │
       ▼ Gitee API 写入
共享仓库 AnnouncementManager / 你的项目名 / user-stats.json
```

### 同步数据格式

```json
{
  "version": "1.0.0",
  "lastUpdated": "2026-06-08T03:23:39.874Z",
  "today": {
    "date": "2026-06-08",
    "userCount": 42
  },
  "dailyStats": [
    { "date": "2026-06-08", "userCount": 42, "updatedAt": "..." },
    { "date": "2026-06-07", "userCount": 38, "updatedAt": "..." }
  ]
}
```

### 为什么用 Gitee 同步？

- **跨实例汇总**：如果你有多个部署实例（比如公网 + 内网），各实例各自统计，但 Gitee 上能看到每个实例的最新数据
- **数据持久化**：即使服务器重装，统计数据仍在 Gitee 仓库中保留最近 30 天
- **透明可查**：直接在 Gitee 上就能看到 JSON 文件内容

### Gitee API 工作方式

1. **首次写入**：文件不存在 → `POST` 创建文件
2. **后续更新**：文件已存在 → 先 `GET` 获取文件 SHA → 再 `PUT` 更新（需要 SHA 参数）
3. 认证方式：URL 查询参数 `?access_token=xxx`

## 常见问题

### Q: 局域网内多台设备访问，算几个用户？

**1 个**。`normalizeIp()` 会把 `192.168.*`、`10.*`、`172.16-31.*` 统一标准化为 `local`，同一局域网内不同设备的 IP 会被合并。但如果 User-Agent 不同（比如一个用 Chrome、一个用 Firefox），则会算作 2 个用户。

### Q: 同一台设备刷新页面，会重复计数吗？

**不会**。同一设备的 IP + UA 不变，哈希值相同，`COUNT(DISTINCT user_hash)` 天然去重。

### Q: 本地开发时统计数据会污染生产环境吗？

会。本地 `127.0.0.1` 也被记录。如果需要在本地开发时禁用统计，启动时不调用 `startScheduler()` 即可；前端请求 `/api/stats/daily` 返回 0 也不影响功能。

### Q: 如何修改统计间隔？

修改 `startScheduler()` 的参数：

```js
startScheduler(30 * 60 * 1000);  // 30分钟
// 或传对象
startScheduler({ intervalMs: 30 * 60 * 1000, enableGiteeSync: false });
```

### Q: 不想同步到 Gitee 怎么办？

```js
startScheduler({ intervalMs: 2 * 60 * 60 * 1000, enableGiteeSync: false });
```

本地统计功能仍然正常，只是不上传到 Gitee。

### Q: 多个项目的统计数据会在 Gitee 上冲突吗？

**不会**。每个项目写入各自子目录下的 `user-stats.json`，互不影响：

```
AnnouncementManager/
```
